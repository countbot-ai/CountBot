# 五、渠道与消息系统

渠道系统让 CountBot 能够同时连接多个聊天平台。消息系统提供可靠的消息投递保障。

## 2、BaseChannel（频道基类）

核心文件：`backend/modules/channels/base.py`

### （1）数据结构

```python
@dataclass
class InboundMessage:
    """入站消息 - 从聊天平台接收"""
    channel: str           # 渠道名（如 "telegram"）
    sender_id: str         # 发送者 ID
    chat_id: str           # 聊天 ID
    content: str           # 消息内容
    media: list[str] = []  # 媒体文件路径列表
    metadata: dict = {}    # 元数据

@dataclass
class OutboundMessage:
    """出站消息 - 发送到聊天平台"""
    channel: str           # 目标渠道
    chat_id: str           # 目标聊天 ID
    content: str           # 消息内容
    media: list[str] = []  # 媒体文件列表
    metadata: dict = {}    # 元数据
```

所有渠道都使用统一的消息格式，这是多渠道统一的基础。

### （2）抽象接口

```python
class BaseChannel(ABC):
    name: str = "base"

    def __init__(self, config):
        self.config = config
        self._running = False
        self._message_callback = None

    @abstractmethod
    async def start(self) -> None:
        """启动频道"""

    @abstractmethod
    async def stop(self) -> None:
        """停止频道"""

    @abstractmethod
    async def send(self, msg: OutboundMessage) -> None:
        """发送消息"""

    @abstractmethod
    async def test_connection(self) -> dict:
        """测试连接"""
```

### （3）权限校验

```python
# base.py:98-114
def is_allowed(self, sender_id):
    """检查发送者是否在白名单中"""
    allow_list = getattr(self.config, "allow_from", [])
    if not allow_list:
        return True  # 空白名单 = 允许所有人

    sender_str = str(sender_id)
    if sender_str in allow_list:
        return True

    # 支持 "id|username" 复合格式
    if "|" in sender_str:
        for part in sender_str.split("|"):
            if part and part in allow_list:
                return True
    return False
```

### （4）入站消息处理

```python
# base.py:120-148
async def _handle_message(self, sender_id, chat_id, content, media=None, metadata=None):
    """处理入站消息：校验权限后转发到回调"""
    # 1. 权限检查
    if not self.is_allowed(sender_id):
        logger.warning(f"Access denied for {sender_id}")
        return

    # 2. 创建统一格式的消息对象
    msg = InboundMessage(
        channel=self.name,
        sender_id=str(sender_id),
        chat_id=str(chat_id),
        content=content,
        media=media or [],
        metadata=metadata or {},
    )

    # 3. 通过回调函数转发给 ChannelManager
    if self._message_callback:
        await self._message_callback(msg)
```

### （2）频道初始化

```python
# manager.py:47-71
def _init_channels(self):
    for name, (module_path, class_name) in _CHANNEL_REGISTRY.items():
        channel_cfg = getattr(channels_config, name, None)
        if not channel_cfg or not getattr(channel_cfg, "enabled", False):
            continue  # 未启用则跳过

        # 动态导入并实例化
        module = __import__(module_path, fromlist=[class_name])
        cls = getattr(module, class_name)
        self.channels[name] = cls(channel_cfg)

    # 为所有频道设置统一的消息回调
    for channel in self.channels.values():
        channel.set_message_callback(self._on_inbound_message)
```

### （3）频道监督与自动重连

```python
# manager.py:104-133
async def _start_channel_supervised(self, name, channel):
    """监督循环 - 异常退出后自动重连（指数退避）"""
    initial_backoff = 5     # 初始等待 5 秒
    max_backoff = 300       # 最大等待 300 秒（5 分钟）
    backoff = initial_backoff

    while self._running:
        start_time = asyncio.get_event_loop().time()
        try:
            await channel.start()   # 启动频道
        except asyncio.CancelledError:
            break
        except Exception as e:
            logger.error(f"Channel {name} error: {e}")

        if not self._running:
            break

        # 如果运行超过 60 秒才断开，重置退避时间
        elapsed = asyncio.get_event_loop().time() - start_time
        if elapsed > 60:
            backoff = initial_backoff

        # 指数退避重连
        await asyncio.sleep(backoff)
        backoff = min(backoff * 2, max_backoff)  # 5 → 10 → 20 → 40 → ... → 300
```

这个设计非常巧妙：
- 如果频道正常运行了很长时间后断开（如网络抖动），退避时间重置为 5 秒，快速重连
- 如果频道反复快速崩溃（如配置错误），退避时间逐步增长，避免频繁重试

### （4）消息路由

```python
# 入站消息路由：频道 → 消息总线
async def _on_inbound_message(self, msg: InboundMessage):
    await self.bus.publish_inbound(msg)

# 出站消息路由：消息总线 → 频道
async def _dispatch_outbound(self):
    while self._running:
        msg = await self.bus.consume_outbound()
        channel = self.channels.get(msg.channel)
        if channel:
            await channel.send(msg)
```

## 4、EnterpriseMessageQueue（企业级消息队列）

核心文件：`backend/modules/messaging/enterprise_queue.py`

### （1）核心特性

```
EnterpriseMessageQueue
├── 四级优先级队列
│   ├── URGENT (3) - 紧急消息
│   ├── HIGH   (2) - 高优先级
│   ├── NORMAL (1) - 默认
│   └── LOW    (0) - 低优先级
├── 消息去重（基于内容哈希）
├── 入站/出站分离
├── 死信队列（DLQ）
├── 自动重试（最多 3 次，降级优先级）
└── 监控指标
```

### （2）优先级调度

```python
# enterprise_queue.py:177-207
async def dequeue(self):
    """按优先级出队"""
    # 先检查高优先级队列
    for priority in [URGENT, HIGH, NORMAL, LOW]:
        queue = self._queues[priority]
        if not queue.empty():
            return await queue.get()

    # 所有队列都空，等待任意队列有消息
    tasks = [asyncio.create_task(queue.get()) for queue in self._queues.values()]
    done, pending = await asyncio.wait(tasks, return_when=asyncio.FIRST_COMPLETED)
    for task in pending:
        task.cancel()
    return done.pop().result()
```

### （3）消息去重

```python
# enterprise_queue.py:251-270
def _hash_message(self, msg):
    """计算消息哈希"""
    content = f"{msg.channel}:{msg.chat_id}:{msg.sender_id}:{msg.content}"
    return hashlib.md5(content.encode()).hexdigest()

def _is_duplicate(self, msg_hash):
    """滑动窗口去重"""
    if msg_hash not in self._message_hashes:
        return False
    timestamp = self._message_hashes[msg_hash]
    if time.time() - timestamp > self._dedup_window:  # 默认 60 秒
        del self._message_hashes[msg_hash]
        return False
    return True
```

去重机制防止同一条消息被处理多次（如网络重传导致重复）。

### （4）失败重试与死信队列

```python
# enterprise_queue.py:236-249
async def mark_failed(self, msg, error):
    msg.retry_count += 1
    if msg.retry_count < msg.max_retries:
        # 降级优先级后重新入队
        lower_priority = MessagePriority(max(0, msg.priority.value - 1))
        await self._queues[lower_priority].put(msg)
    else:
        # 超过最大重试次数 → 进入死信队列
        await self._dead_letter_queue.put((msg, error))
```

重试策略：失败后降低优先级重试，确保不会因为一条消息的失败阻塞其他消息。

## 5、ChannelMessageHandler（频道消息处理器）

核心文件：`backend/modules/channels/handler.py`

### （1）消息处理主循环

```python
# handler.py:159-185
async def start_processing(self):
    """从消息总线消费入站消息"""
    consecutive_errors = 0
    while True:
        try:
            msg = await self.bus.consume_inbound()
            consecutive_errors = 0
            # 每条消息在独立的 asyncio Task 中处理
            asyncio.create_task(self.handle_message(msg))
        except Exception as e:
            consecutive_errors += 1
            if consecutive_errors >= 10:
                await asyncio.sleep(5)  # 连续错误过多，暂停恢复
```

注意：`asyncio.create_task()` 使每条消息并行处理，不会阻塞队列消费。

### （2）单条消息处理流程

```python
# handler.py:187-278
async def handle_message(self, msg):
    # 1. 清理 @mention 标记
    content = _AT_MENTION_RE.sub("", msg.content).strip()

    # 2. 限流检查
    if self.rate_limiter:
        allowed, error_msg = await self.rate_limiter.check(msg.sender_id)
        if not allowed:
            await self._send_reply(msg, error_msg)
            return

    # 3. 命令分发
    cmd = content.lower()
    if cmd in ("/new", "/newsession"):
        await self._handle_new_session_command(msg)
        return
    if cmd in ("/help", "/h", "/?"):
        await self._handle_help_command(msg)
        return
    # ... 其他命令

    # 4. 获取/创建会话
    session_id = await self._get_or_create_session(msg)

    # 5. 保存用户消息到数据库
    await self._save_message(session_id, "user", msg.content)

    # 6. 加载历史消息
    history = await self._get_session_history(session_id)

    # 7. Agent 处理
    response = await self._process_with_agent(
        session_id, msg.content, history, cancel_token,
        channel=msg.channel, chat_id=msg.chat_id,
    )

    # 8. 保存助手回复 + 发送
    if response:
        await self._save_message(session_id, "assistant", response)
        await self._send_reply(msg, response)
```

## 6、RateLimiter（限流器）

核心文件：`backend/modules/messaging/rate_limiter.py`

采用**令牌桶算法**：
- 默认：10 次请求 / 60 秒
- 每个用户独立的令牌桶
- 令牌以恒定速率自动补充

## 7、各渠道实现特点

### （1）飞书（Feishu）

- 使用 WebSocket 长连接
- 通过 multiprocessing 运行独立子进程（避免事件循环冲突）
- 支持文本、图片、文件
- Markdown + 表格卡片富文本输出

### （2）钉钉（DingTalk）

- Stream 模式连接
- 消息重放支持
- 卡片消息格式

### （3）QQ

- 使用官方 qq-botpy SDK
- 可选腾讯云 OSS 媒体上传

### （4）Telegram

- Long polling 模式（无需 webhook）
- 支持自定义代理
